// feature doesn't need (e.g. the stale-data watchdog only applies to BLE-backed
// sensor features — drop CancelID/.staleDataTimedOut and `staleDataWindow`
// entirely if the feature has no streaming hardware source).

import ComposableArchitecture
import Foundation

@Reducer
struct <Name>Feature {
    // Tunables live here as `static let`, not as scattered magic numbers.
    // static let historyWindow: TimeInterval = 3600
    //
    // How long the value may go without a reading before it clears to "--".
    // Readings, not connection state, drive the clear: a link that flaps between
    // .reconnecting and .active says nothing about whether data is arriving.
    // Size it to outlast the sensor's longest silence on an unbroken link
    // (see CadenceFeature.staleDataWindow for how that was worked out).
    static let staleDataWindow: Duration = .seconds(5)

    @Dependency(\.date.now) var now
    @Dependency(\.<name>Client) var <name>Client   // swap for this feature's real @DependencyClient
    @Dependency(\.continuousClock) var clock

    @ObservableState
    struct State: Equatable {
        var currentValue: Int? = nil
        var connectionState: BLECSCClient.ConnectionState = .disconnected   // swap for the real client's connection-state type
        var pairedPeripheralId: UUID? = nil
    }

    enum Action: Equatable {
        case startListening
        case valueReceived(Double)
        case bleConnectionChanged(BLECSCClient.ConnectionState)
        case staleDataTimedOut
    }

    private enum CancelID { case staleDataTimer }

    var body: some ReducerOf<Self> {
        Reduce { state, action in
            switch action {
            case .startListening:
                return .merge(
                    .run { [<name>Client] send in
                        await <name>Client.startScanning()
                        for await value in <name>Client.<name>() {
                            await send(.valueReceived(value))
                        }
                    },
                    .run { [<name>Client] send in
                        for await connectionState in <name>Client.connectionState(.<name>) {
                            await send(.bleConnectionChanged(connectionState))
                        }
                    }
                )

            case .valueReceived(let value):
                guard value >= 0 else {
                    state.currentValue = nil
                    return .none
                }
                state.currentValue = Int(value.rounded())
                // Each reading restarts the stale-data watchdog. Cancelled explicitly:
                // `cancelInFlight` only cancels once the new task starts, which leaves room
                // for the old timeout to land.
                return .merge(
                    .cancel(id: CancelID.staleDataTimer),
                    .run { send in
                        try await clock.sleep(for: Self.staleDataWindow)
                        await send(.staleDataTimedOut)
                    }
                    .cancellable(id: CancelID.staleDataTimer, cancelInFlight: true)
                )

            case .bleConnectionChanged(let connectionState):
                state.connectionState = connectionState
                switch connectionState {
                case .disconnected:
                    state.currentValue = nil
                    return .cancel(id: CancelID.staleDataTimer)
                case .scanning, .connecting, .connected, .reconnecting, .active:
                    // The stale-data watchdog decides when to clear; a link that flaps
                    // between these says nothing about whether data is arriving.
                    return .none
                }

            case .staleDataTimedOut:
                state.currentValue = nil
                return .none
            }
        }
    }
}
