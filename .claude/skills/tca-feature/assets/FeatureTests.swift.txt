// `expireStaleData` only if this feature has a streaming BLE/hardware source;
// delete them otherwise.

import Testing
import Foundation
import ComposableArchitecture
@testable import Cyclometer

@MainActor
@Suite("<Name>Feature")
struct <Name>FeatureTests {

    nonisolated private static let testDate = Date(timeIntervalSince1970: 1_000_000)

    private func makeStore(
        initialState: <Name>Feature.State = <Name>Feature.State(),
        clock: TestClock<Duration> = TestClock(),
        <name>Client: <Name>Client = .testValue,   // swap for this feature's real @DependencyClient
        date: Date = Self.testDate
    ) -> TestStoreOf<<Name>Feature> {
        TestStore(initialState: initialState) {
            <Name>Feature()
        } withDependencies: {
            $0.date = .constant(date)
            $0.continuousClock = clock
            $0.<name>Client = <name>Client
        }
    }

    /// Every reading arms the stale-data watchdog: an exhaustive test runs it out to end
    /// with nothing in flight.
    private func expireStaleData(_ store: TestStoreOf<<Name>Feature>, _ clock: TestClock<Duration>) async {
        await clock.advance(by: <Name>Feature.staleDataWindow)
        await store.receive(\.staleDataTimedOut) {
            $0.currentValue = nil
        }
    }

    @Test("Start listening subscribes to the value and connection-state streams")
    func startListeningSubscribesToStreams() async {
        let (valueStream, valueContinuation) = AsyncStream<Double>.makeStream()
        let (connStream, connContinuation) = AsyncStream<BLECSCClient.ConnectionState>.makeStream()
        var client = <Name>Client.testValue
        client.<name> = { valueStream }
        client.connectionState = { _ in connStream }

        let clock = TestClock()
        let store = makeStore(clock: clock, <name>Client: client)
        await store.send(.startListening)

        connContinuation.yield(.scanning)
        await store.receive(\.bleConnectionChanged) {
            $0.connectionState = .scanning
        }

        valueContinuation.yield(92)
        await store.receive(\.valueReceived) {
            $0.currentValue = 92
        }

        valueContinuation.finish()
        connContinuation.finish()
        await expireStaleData(store, clock)
        await store.finish()
    }

    @Test("Initial currentValue is nil")
    func initialValueIsNil() async {
        let store = makeStore()
        #expect(store.state.currentValue == nil)
    }

    @Test("Negative value clears currentValue")
    func negativeValueClearsState() async {
        let store = makeStore(initialState: <Name>Feature.State(currentValue: 80))

        // Rejected before the watchdog is armed, so nothing is left in flight.
        await store.send(.valueReceived(-1)) {
            $0.currentValue = nil
        }
    }

    // MARK: - Stale-data watchdog
    // Delete this section if the feature has no streaming hardware source.

    /// A live link with a reading on screen.
    private static var onSensor: <Name>Feature.State {
        <Name>Feature.State(currentValue: 80, connectionState: .active)
    }

    @Test("Readings stopping while the link stays .active clear the value")
    func silentSensorOnALiveLinkClears() async {
        let clock = TestClock()
        let store = makeStore(initialState: Self.onSensor, clock: clock)

        await store.send(.valueReceived(82)) {
            $0.currentValue = 82
        }
        // The link never leaves `.active`: the data's absence is the only signal.
        await expireStaleData(store, clock)
    }

    @Test("Each reading restarts the stale-data window")
    func eachReadingRestartsTheWindow() async {
        let clock = TestClock()
        let store = makeStore(initialState: Self.onSensor, clock: clock)

        // Readings 3 s apart for 9 s: a window counted from the first would fire mid-loop,
        // and its unasserted action would fail the next send. Keep the gap shorter than
        // `staleDataWindow`.
        for value in [82.0, 84, 86] {
            await store.send(.valueReceived(value)) {
                $0.currentValue = Int(value)
            }
            await clock.advance(by: .seconds(3))
        }
        await clock.advance(by: <Name>Feature.staleDataWindow - .seconds(3))
        await store.receive(\.staleDataTimedOut) {
            $0.currentValue = nil
        }
    }

    @Test("A flapping link neither restarts nor cancels the window")
    func flappingLinkDoesNotDriveTheClear() async {
        let clock = TestClock()
        let store = makeStore(initialState: Self.onSensor, clock: clock)

        await store.send(.valueReceived(82)) {
            $0.currentValue = 82
        }
        // A drop and a recovery every second, with no data: neither may restart the
        // window (it would never fire) or cancel it (the last reading would hold).
        for connection: BLECSCClient.ConnectionState in [.reconnecting, .active, .reconnecting] {
            await clock.advance(by: .seconds(1))
            await store.send(.bleConnectionChanged(connection)) {
                $0.connectionState = connection
            }
        }
        await clock.advance(by: <Name>Feature.staleDataWindow - .seconds(3))
        await store.receive(\.staleDataTimedOut) {
            $0.currentValue = nil
        }
    }

    @Test("The first reading after a clear shows again, and re-arms the watchdog")
    func firstReadingAfterAClearShows() async {
        let clock = TestClock()
        let store = makeStore(initialState: Self.onSensor, clock: clock)

        await store.send(.valueReceived(82)) {
            $0.currentValue = 82
        }
        await expireStaleData(store, clock)

        await store.send(.valueReceived(88)) {
            $0.currentValue = 88
        }
        await expireStaleData(store, clock)
    }

    @Test("Terminal disconnect clears the value immediately and disarms the watchdog")
    func terminalDisconnectClearsImmediately() async {
        let clock = TestClock()
        let store = makeStore(initialState: Self.onSensor, clock: clock)

        await store.send(.valueReceived(82)) {
            $0.currentValue = 82
        }
        await store.send(.bleConnectionChanged(.disconnected)) {
            $0.connectionState = .disconnected
            $0.currentValue = nil
        }
        // No .staleDataTimedOut follows — an unasserted receive fails the test.
        await clock.advance(by: <Name>Feature.staleDataWindow)
    }
}
